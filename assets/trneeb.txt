            public Card(int number, char suit)

            public override string ToString() => $"{Number}{Suit}";

        public class Deck
        {
            protected List<Card> _cards;
            private static readonly Random _random = new Random();
            protected List<List<Card>> playerHands = new List<List<Card>>();
            public Deck()
            {
                // 1. Define the possible numbers and characters
                var numbers = Enumerable.Range(2, 13);
                var suits = new[] { 'k', 'd', 's', 'b' };

                // 2. Create a "deck" of all 52 unique cards
                _cards = suits.SelectMany(suit =>
                    numbers.Select(number => new Card(number, suit))
                ).ToList();

                Shuffle();
            }

            private void Shuffle()
            {
                int n = _cards.Count;
                for (int i = n - 1; i > 0; i--)
                {
                    int j = _random.Next(i + 1);
                    // Swap the cards
                    (_cards[j], _cards[i]) = (_cards[i], _cards[j]);
                }
            }

            public List<List<Card>> DealAll()
            {
                // playerHands = new List<List<Card>>();
                for (int i = 0; i < 4; i++)
                {
                    playerHands.Add(new List<Card>());
                }

                for (int i = 0; i < _cards.Count; i++)
                {
                    playerHands[i % 4].Add(_cards[i]);
                   
                }

                return playerHands;
            }
        }
